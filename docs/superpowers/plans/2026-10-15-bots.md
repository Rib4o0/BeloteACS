# Bots (fill a seat, auto-play) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a host fill empty seats with bots while waiting, then play a full game in which the bots bid and play their own turns automatically.

**Architecture:** A bot is an ordinary rostered `Player` with a `bot: true` flag (no cookie, never host, never stale). Two pure modules (`server/bot.ts`, `server/randomBot.ts`) turn a `PublicMatch` into a legal `Action`. A process-wide ticker (`server/rooms/botDriver.ts`) advances one bot action per room per beat, driven by a pure `pumpRoom(room)` so it is unit-testable without timers.

**Tech Stack:** Next.js 16 (App Router, Turbopack), React 19, in-memory `globalThis` stores, Server-Sent Events, Node's built-in test runner (`node --test`, TypeScript type-stripping).

**Spec:** `docs/superpowers/specs/2026-10-15-bots-design.md`

## Global Constraints

- **Node ESM import rule.** Any `server/…` module executed by `node --test` (i.e. with a sibling `.test.ts`) must import engine **values** with a **relative path and explicit `.ts` extension** (e.g. `import { cardValue } from "./game/index.ts"`, `import { chooseAction } from "../bot.ts"`). Extensionless and `@/` imports fail under `node --test`. Type-only imports may be extensionless, but use `.ts` for consistency.
- **App/route/client/`server/actions.ts`** code may use the `@/…` alias (these are not run by `node --test`).
- **Client components** (`components/…`) must import **types only** from the engine (`import type { … } from "@/server/game/index.ts"`), never engine values — type imports are erased and safe in the browser bundle.
- **`_`-prefixed folders under `app/` are private** and are not routed.
- **Presence.** `HEARTBEAT_TIMEOUT_MS = 15_000`: a pre-start player is stale when `now - lastSeen > HEARTBEAT_TIMEOUT_MS`. **Bots are exempt** (they never heartbeat and are always connected).
- **Verification gates.** After every task: `npm test` (all suites) and `npm run typecheck` must pass. Run `npm run build` before the browser task.
- **No git repository** exists in this workspace. Do not run `git` commands; each task ends with a **Checkpoint** step instead of a commit.
- **Do not delete** `server/game/**` — it is the finished engine.
- **Do not change** the engine semantics, the SSE/action routes, or existing behaviour for all-human rooms.

## Review Focus

Input classes / failure modes the spec implies but no task's happy-path test covers. Each is pinned to a test in its owning task (noted in brackets).

1. **Host double-clicks “Add bot”** (or two tabs add at once) — must not create duplicate ids, must not exceed four seats; the extra call returns `full`. [Task 1 test + Task 6]
2. **Every human closes their tab mid-game** — the bots must keep playing, because the ticker is connection-independent and `broadcast` to zero subscribers is a no-op. [Task 5 test]
3. **Everyone passes the auction** — the engine redeals; the bots must keep playing rather than stall on an empty redeal. [Task 5 end-to-end test]
4. **A game reaches 151** — the ticker must stop acting once `match.winner` is set (no post-game moves). [Task 5 test]
5. **A room already full of humans** — `addBot` refuses with `full` and does not mutate the roster. [Task 1 test]

```ts
import type { Player, Room, Team } from "../definitions.ts";
import { MAX_PLAYERS } from "./constants.ts";

export type AddBotResult =
  | { ok: true; player: Player }
  | { ok: false; reason: "not_host" | "full" | "started" };

/** True when the id belongs to a server-controlled bot in this room. */
export function isBot(room: Room, playerId: string): boolean {
  return room.game.players.some(
    (player) => player.id === playerId && player.bot === true,
  );
}

/** The smallest index whose id is free in this room. */
function nextBotIndex(room: Room): number {
  let index = 1;
  while (
    room.game.players.some((player) => player.id === `bot:${room.code}:${index}`)
  ) {
    index += 1;
  }
  return index;
}

/** The team with fewer players right now; ties go to A. */
function emptierTeam(room: Room): Team {
  const a = room.game.players.filter((player) => player.team === "A").length;
  const b = room.game.players.filter((player) => player.team === "B").length;
  return a <= b ? "A" : "B";
}

/**
 * Adds a server-controlled bot to the next empty seat, on the emptier team.
 * Mirrors `startGame(room, actorId)`: only the host may add a bot.
 */
export function addBot(room: Room, actorId: string): AddBotResult {
  const host = room.game.players.find((player) => player.host);

  if (!host || host.id !== actorId) {
    return { ok: false, reason: "not_host" };
  }

  if (room.game.playing) {
    return { ok: false, reason: "started" };
  }

  if (room.game.players.length >= MAX_PLAYERS) {
    return { ok: false, reason: "full" };
  }

  const index = nextBotIndex(room);
  const player: Player = {
    id: `bot:${room.code}:${index}`,
    name: `Bot ${index}`,
    host: false,
    team: emptierTeam(room),
    bot: true,
  };

  room.game.players.push(player);
  return { ok: true, player };
}
```

```ts
import type { Action, PublicMatch, Rng } from "./game/index.ts";

/**
 * Picks a uniformly random legal action. Returns null when the viewer has no
 * legal action (not their turn, the game is over, or they are a spectator).
 */
export function chooseAction(
  view: PublicMatch,
  rng: Rng = Math.random,
): Action | null {
  if (view.legal.length === 0) {
    return null;
  }

  const index = Math.min(
    view.legal.length - 1,
    Math.floor(rng() * view.legal.length),
  );
  return view.legal[index];
}
```

```ts
const SEATS: [string, string, string, string] = ["p0", "p1", "p2", "p3"];
const card = (suit: Card["suit"], rank: Card["rank"]): Card => ({ suit, rank });

/** A fresh match with seat 1 on turn holding a hand the test controls. */
function biddingAt(hand: Card[], top: Contract | null = null): Match {
  const match = createMatch(SEATS, seededRng(1));
  const bidding = match.hand as Extract<Match["hand"], { phase: "bidding" }>;
  bidding.hands = { 0: [], 1: hand, 2: [], 3: [] };
  bidding.rest = { 0: [], 1: [], 2: [], 3: [] };
  bidding.top = top;
  bidding.turn = 1;
  return match;
}

/** A match in trick play with seat 1 on turn and the given table/hand. */
function playingAt(
  hand: Card[],
  trick: { seat: Seat; card: Card }[],
  contract: Contract = { kind: "suit", suit: "hearts" },
): Match {
  const match = createMatch(SEATS, seededRng(1));
  const playing: PlayingHand = {
    phase: "playing",
    dealer: 0,
    turn: 1,
    hands: { 0: [], 1: hand, 2: [], 3: [] },
    contract,
    taker: 1,
    trick,
    tricks: [],
  };
  match.hand = playing;
  return match;
}

function choose(match: Match, seatId = "p1"): Action | null {
  return chooseAction(publicState(match, seatId));
}

test("bids the strongest suit on a strong hand", () => {
  const match = biddingAt([card("hearts", "J"), card("hearts", "9"), card("hearts", "A"), card("hearts", "10"), card("hearts", "K")]);

  const action = choose(match);

  assert.equal(action?.kind, "bid");
  if (action?.kind === "bid") assert.equal(action.contract.kind === "suit" && action.contract.suit, "hearts");
});

test("passes on a worthless hand", () => {
  const match = biddingAt([card("clubs", "7"), card("diamonds", "8"), card("hearts", "7"), card("spades", "8"), card("diamonds", "7")]);

  assert.equal(choose(match)?.kind, "pass");
});

test("leads a plain ace when it holds one", () => {
  const match = playingAt([card("clubs", "A"), card("diamonds", "7"), card("diamonds", "8"), card("diamonds", "9"), card("spades", "10")], []);

  const action = choose(match);

  assert.equal(action?.kind, "play");

});

    assert.equal(result.error, undefined, `engine rejected ${JSON.stringify(action)}`);
```

```ts
const BID_MIN_SCORE = 40;

/** The bot's cards valued as if `suit` were trump, so J/9 dominate. */
function suitScore(hand: readonly Card[], suit: Suit): number {
  const contract: Contract = { kind: "suit", suit };
  return hand.reduce((total, card) => total + cardValue(card, contract), 0);
}

/** Lowest by trick order, ties broken by point value. */
function pickLowest(actions: PlayAction[], contract: Contract): PlayAction {
  return actions.reduce((best, action) => {
    const order = trickOrder(action.card, contract);
    const bestOrder = trickOrder(best.card, contract);
    if (order !== bestOrder) return order < bestOrder ? action : best;
    return cardValue(action.card, contract) < cardValue(best.card, contract)
      ? action
      : best;
  });
}

/** Lowest by point value (a discard, not a win). */
function pickCheapest(actions: PlayAction[], contract: Contract): PlayAction {
  return actions.reduce((best, action) =>
    cardValue(action.card, contract) < cardValue(best.card, contract) ? action : best,
  );
}

function chooseBid(view: PublicMatch): Action | null {
  const bids = view.legal.filter(
    (action): action is BidAction => action.kind === "bid",
  );

  let best: BidAction | null = null;
  let bestScore = 0;

  for (const bid of bids) {
    if (bid.contract.kind !== "suit") continue;
    const score = suitScore(view.hand.hand, bid.contract.suit);
    if (score > bestScore) {
      bestScore = score;
      best = bid;
    }
  }

  if (best && bestScore >= BID_MIN_SCORE) {
    return best;
  }

  return view.legal.find((action) => action.kind === "pass") ?? null;
}

function choosePlay(view: PublicMatch): Action | null {
  const plays = view.legal.filter(
    (action): action is PlayAction => action.kind === "play",
  );

  const you = view.you;
  const contract = view.hand.contract;

  if (plays.length === 0 || you === null || !contract) {
    return null;
  }

  const trick = view.hand.trick;

  if (trick.length === 0) {
    const plainAce = plays.find(
      (action) => !isTrumpCard(action.card, contract) && action.card.rank === "A",
    );
    if (plainAce) return plainAce;

    const plain = plays.filter((action) => !isTrumpCard(action.card, contract));
    return pickLowest(plain.length > 0 ? plain : plays, contract);
  }

  if (trickWinner(trick, contract) === partnerSeat(you)) {
    return pickCheapest(plays, contract);
  }

  const winners = plays.filter(
    (action) =>
      trickWinner([...trick, { seat: you, card: action.card }], contract) === you,
  );

  return winners.length > 0
    ? pickLowest(winners, contract)
    : pickCheapest(plays, contract);
}

/**
 * A simple, deterministic legal move: open on the best suit when worth it,
 * otherwise pass; lead a plain ace or low; when following, discard under a
 * winning partner and beat an opponent as cheaply as possible.
 */
export function chooseAction(view: PublicMatch): Action | null {
  if (view.legal.length === 0) {
    return null;
  }

  return view.hand.phase === "bidding" ? chooseBid(view) : choosePlay(view);
}
```

```ts
import { publicState } from "../game/index.ts";
import { chooseAction } from "../bot.ts";
import { applyGameAction, fetchRooms } from "../Rooms.ts";
import { broadcast } from "./broadcast.ts";
import { isBot } from "./bots.ts";
import type { Room } from "../definitions.ts";

/** How long between bot moves (one move per room per tick). */
export const TICK_MS = 700;

const globalForDriver = globalThis as unknown as {
  botDriverTimer?: ReturnType<typeof setInterval>;
};

/**
 * If the seat on turn belongs to a bot, applies one legal bot action and
 * broadcasts. Returns whether it acted. Timer-free, so it is unit-testable.
 */
export function pumpRoom(room: Room): boolean {
  const match = room.match;

  if (!match || match.winner) {
    return false;
  }

  const playerId = match.seats[match.hand.turn];

  if (!isBot(room, playerId)) {
    return false;
  }

  const action = chooseAction(publicState(match, playerId));

  if (!action) {
    return false;
  }

  const result = applyGameAction(room, action);

  if (!result.ok) {
    return false;
  }

  broadcast(room);
  return true;
}

/**
 * Registers the process-wide ticker once. Every tick advances at most one
 * action per room (so bot moves are watchable) and runs regardless of who is
 * connected. Safe to call repeatedly.
 */
export function ensureBotDriver(): void {
  if (globalForDriver.botDriverTimer) {
    return;
  }

  globalForDriver.botDriverTimer = setInterval(() => {
    for (const room of fetchRooms()) {
      pumpRoom(room);
    }
  }, TICK_MS);
}

/** Cancels the ticker (used by tests and shutdown). */
export function stopBotDriver(): void {
  if (globalForDriver.botDriverTimer) {
    clearInterval(globalForDriver.botDriverTimer);
    globalForDriver.botDriverTimer = undefined;
  }
}

/** Whether the ticker is currently registered. */
export function botDriverRunning(): boolean {
  return globalForDriver.botDriverTimer !== undefined;
}
```

```ts
  _prevState: AddBotState,
```
